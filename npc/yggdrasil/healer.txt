-	script	Healer	FAKE_NPC,{
	.@price = 100;		// Zeny necessário para curar, grátis para VIPs
	.@Delay = 30;		// Tempo de espera para jogadores normais (segundos)
	.@VIPDelay = 5;		// Tempo de espera para VIPs (segundos)
	.@isVIP = getgroupid() > 0;  // VIP se o Group ID for maior que 0 (Normal = 0, VIP = 1+, GM = 99)

	.@resta = @HD - gettimetick(2);
	if (@HD > gettimetick(2)) {
		npctalk "Aguarde " + .@resta + " segundos para ser atendido novamente!";
		end;
	}

	// Se o jogador for VIP, ele não paga
	if (!.@isVIP && .@price) {
		npctalk "[Enfermeira] O tratamento custa " + .@price + " Zeny.";
		if (Zeny < .@price) {
			npctalk "Você não tem Zeny suficiente!";
			end;
		}
		Zeny -= .@price;
	}

	if (.@isVIP) {
		// Mensagem especial para VIPs
		setarray .@vipMsg$[
			0], "Seu plano de saúde cobre esse tratamento, aproveite!",
			"Atendimento VIP sempre garantido, senhor!",
			"Ah, um cliente VIP! Pode sentar que eu cuido de tudo.",
			"Nada melhor que tratamento premium, não é mesmo?",
			"Só os melhores têm esse privilégio!";

		npctalk .@vipMsg$[rand(5)];
	} else {
		// Lista de falas normais
		setarray .@frases$[
			0], "Isso pode arder um pouquinho...",
			"Vou curar, mas toma banho de vez em quando, tá?",
			"Cuidado para não se machucar de novo, bobão!",
			"Pronto! Agora pode apanhar mais um pouco.",
			"Vou cobrar da próxima vez, hein!";

		// Escolhe uma frase aleatória e fala no NPC
		npctalk .@frases$[rand(5)];
	}

	// Efeito visual de cura + cura total
	specialeffect EF_HEAL2, AREA, playerattached();
	percentheal 100,100;

	if (.@isVIP) {
		// Aplica buffs para jogadores VIPs
		specialeffect(EF_INCAGILITY, AREA, playerattached());
		sc_start SC_INC_AGI, 300000, 10;
		specialeffect(EF_BLESSING, AREA, playerattached());
		sc_start SC_BLESSING, 300000, 10;
	}

	// Define o tempo de espera antes do próximo uso, conforme o tipo de jogador
	@HD = gettimetick(2) + (.@isVIP ? .@VIPDelay : .@Delay);

	end;
}
